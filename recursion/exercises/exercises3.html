Ejercicio 1 Fácil

Contar Ocurrencias de Carácter

Escribe contarCaracter(str, c) que cuente cuántas veces aparece el carácter c en str.

Ejemplo:

Entrada: contarCaracter("banana", "a")

Salida: 3

Caso base: str vacío retorna 0. Compara str[0] con c, suma 1 si coincide, luego recursión con slice(1).

function contarCaracter(str, c) {
    if (str.length === 0) return 0;
    
    const coincide = (str[0] === c) ? 1 : 0;
    return coincide + contarCaracter(str.slice(1), c);
}

// Ejemplos
console.log(contarCaracter("banana", "a")); // 3
console.log(contarCaracter("hello", "l"));  // 2
console.log(contarCaracter("", "x"));       // 0
Ejercicio 2 Fácil

Remover Todos los Espacios

Crea removerEspacios(str) que elimine todos los espacios en blanco de un string.

Ejemplo:

Entrada: removerEspacios("Hola mundo desde JS")

Salida: "HolamundodesdeJS"

Similar a removerCaracter pero con ' ' como carácter a eliminar.

function removerEspacios(str) {
    if (str.length === 0) return "";
    
    const primero = str[0];
    const resto = removerEspacios(str.slice(1));
    
    return (primero === ' ') ? resto : primero + resto;
}

// Ejemplos
console.log(removerEspacios("Hola mundo desde JS")); // "HolamundodesdeJS"
console.log(removerEspacios("   espacios   "));      // "espacios"
console.log(removerEspacios(""));                    // ""
Ejercicio 3 Fácil

Convertir a Mayúsculas Recursivo

Implementa aMayusculas(str) que convierta todas las letras a mayúsculas sin usar toUpperCase() en el string completo.

Ejemplo:

Entrada: aMayusculas("hola mundo")

Salida: "HOLA MUNDO"

Procesa un carácter a la vez. Usa charCodeAt y manipulación de códigos ASCII, o simplemente llama a toUpperCase() en cada carácter.

function aMayusculas(str) {
    if (str.length === 0) return "";
    
    const primero = str[0].toUpperCase();
    const resto = aMayusculas(str.slice(1));
    
    return primero + resto;
}

// Ejemplos
console.log(aMayusculas("hola"));     // "HOLA"
console.log(aMayusculas("Hello World")); // "HELLO WORLD"
console.log(aMayusculas(""));         // ""
Ejercicio 4 Fácil

Contar Dígitos en String

Crea contarDigitos(str) que cuente cuántos caracteres son dígitos numéricos (0-9).

Ejemplo:

Entrada: contarDigitos("a1b2c3d4")

Salida: 4

Verifica si str[0] está entre '0' y '9'. Puedes comparar códigos ASCII o usar regex.

function contarDigitos(str) {
    if (str.length === 0) return 0;
    
    const esDigito = (str[0] >= '0' && str[0] <= '9') ? 1 : 0;
    return esDigito + contarDigitos(str.slice(1));
}

// Ejemplos
console.log(contarDigitos("a1b2c3d4")); // 4
console.log(contarDigitos("12345"));    // 5
console.log(contarDigitos("abc"));      // 0
Ejercicio 5 Fácil

Reemplazar Vocales

Reemplaza todas las vocales por un carácter específico (ej: '*').

Ejemplo:

Entrada: reemplazarVocales("hola mundo", "*")

Salida: "h*l* m*nd*"

Define un string de vocales "aeiouAEIOU". Si str[0] es vocal, reemplaza; si no, mantén.

function reemplazarVocales(str, reemplazo) {
    const vocales = 'aeiouAEIOU';
    
    if (str.length === 0) return "";
    
    const primero = str[0];
    const resto = reemplazarVocales(str.slice(1), reemplazo);
    
    const nuevoChar = vocales.includes(primero) ? reemplazo : primero;
    return nuevoChar + resto;
}

// Ejemplos
console.log(reemplazarVocales("hola mundo", "*")); // "h*l* m*nd*"
console.log(reemplazarVocales("JavaScript", "#")); // "J#v#Scr#pt"
console.log(reemplazarVocales("xyz", "*"));        // "xyz"
Ejercicio 6 Medio

Capitalizar Primera Letra

Crea capitalizarPrimera(str) que solo convierta a mayúscula la primera letra del string.

Ejemplo:

Entrada: capitalizarPrimera("hola mundo")

Salida: "Hola mundo"

Caso base: string vacío. Procesa primer carácter a mayúscula, el resto queda igual.

function capitalizarPrimera(str) {
    if (str.length === 0) return "";
    
    const primera = str[0].toUpperCase();
    const resto = str.slice(1); // No recursivo para el resto
    
    return primera + resto;
}

// Versión completamente recursiva (menos eficiente)
function capitalizarPrimeraRec(str) {
    if (str.length === 0) return "";
    
    const primera = str[0].toUpperCase();
    const resto = capitalizarPrimeraRec(str.slice(1));
    
    return primera + resto;
}

// Ejemplos
console.log(capitalizarPrimera("hola"));    // "Hola"
console.log(capitalizarPrimera(""));        // ""
console.log(capitalizarPrimera("a"));       // "A"
Ejercicio 7 Medio

Eliminar Duplicados Consecutivos

Remueve caracteres que se repiten seguidos: "aaabbb" → "ab".

Ejemplo:

Entrada: eliminarDuplicados("aabbccdd")

Salida: "abcd"

Caso base: ≤1 carácter. Compara str[0] con str[1]. Si iguales, omite str[0].

function eliminarDuplicados(str) {
    if (str.length <= 1) return str;
    
    const primero = str[0];
    const segundo = str[1];
    const resto = str.slice(1);
    
    // Si el primer y segundo carácter son iguales, omitir el primero
    if (primero === segundo) {
        return eliminarDuplicados(resto);
    } else {
        return primero + eliminarDuplicados(resto);
    }
}

// Ejemplos
console.log(eliminarDuplicados("aaabbbccc")); // "abc"
console.log(eliminarDuplicados("Hello"));      // "Helo"
console.log(eliminarDuplicados("a"));          // "a"
console.log(eliminarDuplicados(""));           // ""
Ejercicio 8 Medio

Reverse String con Índice

Implementa reversaIndice(str, indice) que revierta el string usando un índice en lugar de slice.

Ejemplo:

Entrada: reversaIndice("hello", str.length-1)

Salida: "olleh"

Accede al carácter en indice. Caso base: indice < 0. Paso: str[indice] + reversaIndice(indice-1).

function reversaIndice(str, indice) {
    // Caso base
    if (indice < 0) {
        return "";
    }
    
    // Carácter actual + reversa del resto
    return str[indice] + reversaIndice(str, indice - 1);
}

function reversa(str) {
    return reversaIndice(str, str.length - 1);
}

// Ejemplos
console.log(reversa("hello"));     // "olleh"
console.log(reversa("recursión")); // "noisrucer"
console.log(reversa(""));          // ""
Ejercicio 9 Medio

String sin Vocales

Crea sinVocales(str) que retorne el string sin ninguna vocal.

Ejemplo:

Entrada: sinVocales("hello world")

Salida: "hll wrld"

Similar a removerCaracter pero con conjunto de vocales.

function sinVocales(str) {
    const vocales = 'aeiouAEIOU';
    
    if (str.length === 0) return "";
    
    const primero = str[0];
    const resto = sinVocales(str.slice(1));
    
    // Si es vocal, omitir; si no, conservar
    return vocales.includes(primero) ? resto : primero + resto;
}

// Ejemplos
console.log(sinVocales("hello world")); // "hll wrld"
console.log(sinVocales("JavaScript"));   // "JvScrpt"
console.log(sinVocales("AEIOUaeiou"));   // ""
Ejercicio 10 Difícil

Expandir Caracteres de Repetición

Dado un string como "a3b2c1", expande las repeticiones: "aaabbc".

Ejemplo:

Entrada: expandir("a3b2c1")

Salida: "aaabbc"

Necesitas procesar de 2 en 2: carácter + número. Usa parseInt() o Number() para convertir.

function expandir(str) {
    // Caso base: string vacío o longitud 1 (incompleto)
    if (str.length < 2) return "";
    
    const char = str[0];
    const numStr = str[1];
    const resto = str.slice(2);
    
    // Convertir dígito a número
    const veces = parseInt(numStr, 10);
    
    // Construir string repetido
    let repetido = "";
    for (let i = 0; i < veces; i++) {
        repetido += char;
    }
    
    // Recursión
    return repetido + expandir(resto);
}

// Versión que maneja números de más de un dígito
function expandirAvanzado(str) {
    if (str.length === 0) return "";
    
    // Extraer carácter
    const char = str[0];
    
    // Extraer número (puede ser múltiples dígitos)
    let i = 1;
    let numStr = "";
    while (i < str.length && str[i] >= '0' && str[i] <= '9') {
        numStr += str[i];
        i++;
    }
    
    const veces = parseInt(numStr, 10);
    const resto = str.slice(i);
    
    let repetido = "";
    for (let j = 0; j < veces; j++) {
        repetido += char;
    }
    
    return repetido + expandirAvanzado(resto);
}

// Ejemplos
console.log(expandir("a3b2c1"));     // "aaabbc"
console.log(expandir("x5y1z2"));     // "xxxxxyz z"? No: "xxxxxyz z" error
console.log(expandirAvanzado("a10b2")); // "aaaaaaaaaab b"? "aaaaaaaaaabb"
